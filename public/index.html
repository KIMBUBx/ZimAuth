Python Client Example 
import requests

# Point to your ZimAuth server
BASE_URL = "http://localhost:3000/api/1.2/"

# 1. Initialize application
resp = requests.post(BASE_URL, data={
    "type": "init",
    "name": "MyCheatOrApp",
    "ownerid": "YOUR_OWNER_ID",
    "secret": "YOUR_APP_SECRET",
    "version": "1.0"
}).json()

sessionid = resp["sessionid"]

# 2. Login with License Key & HWID
login = requests.post(BASE_URL, data={
    "type": "license",
    "key": "ZIM-XXXX-XXXX",
    "hwid": "HWID_12345",
    "sessionid": sessionid
}).json()

if login["success"]:
    print("Authentication Succeeded! Enjoy.")
